生产者-消费者问题#

 1#define N 100  /* 缓冲区个数 */
 2semaphore mutex =1;  /* 互斥信号量：控制对临界区的访问 */
 3semaphore empty =N;  /* 空缓冲区个数 */
 4semaphore full = 0;  /* 满缓冲区个数 */
 5
 6void producer(void){ 
 7	int item;
 8	while(TRUE) {
 9		item=produce_item();
10		P(&empty);
11		P(&mutex);
12		
13		insert_item(item);
14		
15		V(&mutex);
16		V(&full);
17	}
18}
19
20void consumer(void){ 
21	int item;
22	while(TRUE) {
23		P(&full);
24		P(&mutex);
25		
26		item=remove_item();
27		
28		V(&mutex);
29		V(&empty);
30		consume_item(item);
31	}
32}

读者写者问题#

读者和写者两组并发进程共享一个文件，要求：多个读者可以同时读文件；只允许一个写者写文件；任一写者在完成写操作之前不允许其他人工作；写者写之前，应该让已有的读写者都退出。

  • 分析：本质就是连续多个同类进程访问同一个临界资源的问题
  • 2类进程：读者和写者。
  • 其中有互斥关系的是：写者之间、写者与读者之间。

参考：408OS_PV操作大题总结

读优先#

 1
 2int count = 0; // 读者数量
 3
 4semaphore busy = 1; // "读文件"和"写文件"的互斥锁
 5
 6semaphore mutex = 1; // 变量 count 的互斥锁
 7
 8Reader(){ // 读者进程
 9    while(1){
10        P(mutex); // 给count上锁
11        count++;
12        if (count == 1){ // 有人读时加锁，后面人继续读不用再加锁了
13            P(busy); // 有读者时因互斥关系不能写
14        }
15        V(mutex);
16
17        // 读文件;
18
19        P(mutex);
20        count--;
21        if (count == 0){ // 没有读者时去锁
22            V(busy);
23        }
24        V(mutex);
25    }
26}
27
28  
29
30Writer(){ // 写者进程
31    while(1){
32        P(busy);
33        // 写文件;
34        V(busy);
35    }
36}

读优先缺点是读进程会抢占资源导致写者饥饿。

读写公平#

 1
 2int count = 0;
 3semaphore queue = 1;  // 实现"读写公平"的互斥锁，可以视为一个队列
 4semaphore busy = 1;   // "读文件"和"写文件"的互斥锁
 5semaphore mutex = 1;  // 变量 count 的互斥锁
 6
 7Reader(){ // 读者进程
 8    while(1){
 9        P(queue); // 在无写进程请求时不需要进入队列
10        P(mutex); // 该互斥量实际上是多余的，上面语句已经兼有互斥功能
11        count++;
12        if (count == 1){
13            P(busy);
14        }
15        V(mutex); // 该互斥量实际上是多余的，下面语句已经兼有此功能，可以去掉
16        V(queue); // 恢复对共享文件的访问，不然其他读者没法读了
17
18        // 读文件;
19
20        P(mutex);
21        count--;
22        if (count == 0){
23            V(busy);
24        }
25        V(mutex);
26    }
27}
28
29  
30
31Writer(){ // 写者进程
32    while(1){
33        P(queue); // 在无其他写进程请求时不需要进入队列
34        P(busy);
35
36        // 写文件;
37
38        V(busy);
39        V(queue); // 恢复对共享文件的访问
40    }
41}

这样子有一个写者排入队列就会把后面的读者堵住，后面的读者形成不了抢占。

顺序为：之前的读者-写者-后来的读者

写优先#

 1
 2int ReaderCount = 0; // 读者数量
 3int WriterCount = 0; // 写者数量
 4
 5semaphore Read = 1; // "读文件"的互斥锁
 6semaphore Write = 1; // "写文件"的互斥锁
 7semaphore ReaderMutex = 1; // 变量 ReaderCount 的互斥锁
 8semaphore WriterMutex = 1; // 变量 WriterCount 的互斥锁  
 9
10Reader(){ // 读者进程
11    while(1){
12        P(Read); // 每个读进程都需要对 Read 加锁
13        P(ReaderMutex); // 对 ReadCount 的互斥，实际上，上条语句已经兼有此功能，可以去掉
14
15        ReaderCount++;
16
17        if (ReaderCount == 1){ // 如果是第一个读进程
18            P(Write); // 则对写者上锁
19        }
20
21        V(ReaderMutex); // 对 ReadCount 的互斥，实际上，下条语句已经兼有此功能，可以去掉
22        V(Read); // Read 解锁
23
24        读文件;
25
26        P(ReaderMutex); // 对 ReadCount 的互斥
27        ReaderCount--;
28        if (ReaderCount == 0){ // 如果是最后一个读进程
29            V(Write); // 则对写者解锁
30        }
31
32        V(ReaderMutex); // 对 ReadCount 的互斥
33        }
34}
35
36  
37
38Writer(){ // 写者进程
39    while(1){
40        P(WriterMutex); // 对 WriterCount 的互斥
41        
42        WriterCount++;
43        
44        if (WriterCount == 1){ // 如果是第一个写进程
45            P(Read); // 则对读者上锁
46        }
47
48        V(WriterMutex); // 对 WriterCount 的互斥
49
50        P(Write); // Write 加锁
51        写文件;
52        V(Write); // Write 解锁
53
54        P(WriterMutex); // 对 WriterCount 的互斥
55
56        WriterCount--;
57
58        if (WriterCount == 0){ // 如果是最后一个写进程
59            V(Read); // 则对读者解锁
60        }
61
62        V(WriterMutex); // 对 WriterCount 的互斥
63    }
64}

食品供货问题#

某商店有两种食品A和B， 最大数量各为M个。该商店将A、B两种食品搭配出售，每次各取一个。为避免食品变质，遵循先到食品先出售的原则。有两个食品公司分别不断地供应A、B两种食品(每次一个)。为保证正常销售，当某种食品的数量比另一种的数量超过K(K<M)个时，暂停对数量大的食品进货。

这就是两个生产者-消费者关系。

 1int countA = 0, countB = 0; // 食品数量
 2
 3// 信号量
 4semaphore mutex = 1;
 5semaphore emptyA = m, emptyB = m;
 6semaphore foodA = 0, foodB = 0;
 7semaphore limitA = k, limitB = k; // A比B, B比A的富余量
 8
 9// 供应A进程
10ProviderA() {
11    while(true) {
12        P(emptyA);       // 是否还有空货架放A?
13        P(limitB);       // 消耗一个B对A的富余名额
14        
15        P(mutex);
16        countA++;        // 生产一个A并放入
17        V(mutex);
18
19        V(limitA);       // 增加一个A对B的富余名额
20        V(foodA);        // 增加一个食品A
21    }
22}
23
24// 供应B进程
25ProviderB() {
26    while(true) {
27        P(emptyB);       // 是否还有空货架放B?
28        P(limitA);       // 消耗一个A对B的富余名额
29        
30        P(mutex);
31        // 生产一个B并放入
32        countB++;
33        V(mutex);
34
35        V(limitB);       // 增加一个B对A的富余名额
36        V(foodB);        // 增加一个食品B
37    }
38}
39
40// 销售员进程
41Seller() {
42    while(true) {
43        P(foodA); // 是否有食品A?
44        P(foodB); // 是否有食品B?
45
46        P(mutex);
47        // 取走A和B
48        countA--;
49        countB--;
50        V(mutex);
51
52        V(emptyA); // 货架空出
53        V(emptyB); // 货架空出
54    }
55}

三峡大坝船闸调度问题#

由于水面高度不同，有160175米的落差，所以三峡大坝有五级船闸，T1T5。由上游驶来的船需经由各级船闸到下游；由下游驶来的船需经由各级船闸到上游。假设只能允许单方向通行（此假设与实际情况不符，实际为双向各五级船闸）。

这是两个读者。出于简化，我们不考虑5个闸门。

 1// 信号量和计数器
 2semaphore up_mutex = 1, down_mutex = 1;
 3semaphore direction_mutex = 1;
 4int up_count = 0, down_count = 0;
 5
 6// 上行船只进程 
 7Upward_Ship() {
 8    P(up_mutex);
 9    if (up_count == 0) { // 如果是第一艘上行船
10        P(direction_mutex); // 尝试获取航行方向控制权
11    }
12    up_count++;
13    V(up_mutex);
14	
15	cross();  // 通过闸门
16    
17    P(up_mutex);
18    up_count--;
19    if (up_count == 0) { // 如果是最后一艘上行船
20        V(direction_mutex); // 释放航行方向控制权
21    }
22    V(up_mutex);
23}
24
25// 下行船只进程 (上游 -> 下游: T1 -> T5)
26Downward_Ship() {
27    P(down_mutex);
28    if (down_count == 0) { // 如果是第一艘下行船
29        P(direction_mutex); // 尝试获取航行方向控制权
30    }
31    down_count++;
32    V(down_mutex);
33
34    cross();  // 通过闸门
35    
36    P(down_mutex);
37    down_count--;
38    if (down_count == 0) { // 如果是最后一艘下行船
39        V(direction_mutex); // 释放航行方向控制权
40    }
41    V(down_mutex);
42}

狒狒过峡谷问题#

一个主修人类学、辅修计算机科学的学生参加了一个研究课题，调查是否可以教会非洲狒狒理解死锁。他找到一处很深的峡谷，在上边固定了一根横跨峡谷的绳索，这样狒狒就可以攀住绳索越过峡谷。同一时刻，只要朝着相同的方向就可以有几只狒狒通过。但如果向东和向西的狒狒同时攀在绳索上那么会产生死锁（狒狒会被卡在中间），由于它们无法在绳索上从另一只的背上翻过去。如果一只狒狒想越过峡谷，它必须看当前是否有别的狒狒正在逆向通行。利用信号量编写一个避免死锁的程序来解决该问题。不考虑连续东行的狒狒会使得西行的狒狒无限制地等待的情况。

参考：Operating Systems Notes 10: Classic Synchronization Problems

不考虑饥饿#

在不考虑饥饿的情况下，这就是和上面的狒狒过峡谷一样了。

 1// 信号量和计数器
 2semaphore mutex = 1;
 3semaphore rope = 1;
 4int east_count = 0, west_count = 0; // 可以合并为一个方向计数器和一个方向变量
 5
 6// 向东走的狒狒
 7Eastward_Baboon() {
 8    P(mutex);
 9    if (east_count == 0) { // 如果是第一只向东的狒狒
10        P(rope);           // 尝试获取绳索
11    }
12    east_count++;
13    V(mutex);
14
15    // --- 过峡谷 ---
16     crossing();
17    // --- 过完峡谷 ---
18
19    P(mutex);
20    east_count--;
21    if (east_count == 0) { // 如果是最后一只向东的狒狒
22        V(rope);           // 释放绳索
23    }
24    V(mutex);
25}
26
27// 向西走的狒狒
28Westward_Baboon() {
29    P(mutex);
30    // 这里我们用 west_count 来实现, 逻辑完全对称
31    if (west_count == 0) { // 如果是第一只向西的狒狒
32        P(rope);           // 尝试获取绳索
33    }
34    west_count++;
35    V(mutex);
36
37    // --- 过峡谷 ---
38    crossing();
39    // --- 过完峡谷 ---
40
41    P(mutex);
42    west_count--;
43    if (west_count == 0) { // 如果是最后一只向西的狒狒
44        V(rope);           // 释放绳索
45    }
46    V(mutex);
47}

考虑饥饿#

考虑饥饿的话，可以让两边等待的狒狒都去争turn。谁争到谁上。

 1// 信号量和计数器
 2semaphore mutex_e = 1, mutex_w = 1;
 3semaphore rope = 1;
 4semaphore turn = 1; 
 5int east_count = 0, west_count = 0;
 6
 7// 向东走的狒狒
 8Eastward_Baboon() {
 9    P(turn); // 在旋转门前排队
10    P(mutex_e);
11    east_count++;
12    if (east_count == 1) { // 如果是第一只东行狒狒
13        P(rope);           // 尝试为东行方向获取绳索
14    }
15    V(mutex_e);
16    V(turn); // 通过旋转门, 让下一个人/狒狒可以排队
17
18    // --- 过峡谷 ---
19
20    P(mutex_e);
21    east_count--;
22    if (east_count == 0) { // 如果是最后一只东行狒狒
23        V(rope);           // 释放绳索
24    }
25    V(mutex_e);
26}
27
28// 向西走的狒狒
29Westward_Baboon() {
30    P(turn); // 在旋转门前排队
31    P(mutex_w);
32    west_count++;
33    if (west_count == 1) { // 如果是第一只西行狒狒
34        P(rope);           // 尝试为西行方向获取绳索
35    }
36    V(mutex_w);
37    V(turn); // 通过旋转门
38
39    // --- 过峡谷 ---
40
41    P(mutex_w);
42    west_count--;
43    if (west_count == 0) { // 如果是最后一只西行狒狒
44        V(rope);           // 释放绳索
45    }
46    V(mutex_w);
47}

睡眠理发师问题#

也是生产者-消费者模型

 1// 常量和变量
 2#define N 5 // 假设有5把椅子
 3int waiting_chairs = N; // 可用的等候椅数量, 就是生产者-消费者模型中的 empty 信号量
 4
 5// 信号量
 6semaphore customers = 0; // 等待的顾客数, 就是生产者-消费者模型中的 full 信号量
 7semaphore barbers = 0;   // 空闲的理发师数, 加入的新信号量, 因为必须理完发才能走, 所以必须找到一个空闲的理发师
 8semaphore mutex = 1;     // 保护 waiting_chairs, 就是生产者-消费者模型中的 mutex 信号量
 9
10// 顾客进程-生产者
11Customer() {
12    P(mutex);
13    if (waiting_chairs > 0) {
14        waiting_chairs--; // 坐下一把椅子
15        V(mutex);         // 释放锁
16        V(customers);     // 通知理发师, 我来了
17
18        // 如果是生产者-消费者模型, 这里就可以结束了, 因为进入就可以结束了, 但是理发坐下之后还得等
19
20        P(barbers);       // 等待理发师准备好 (即理发师被叫醒给自己理发)
21        
22        // --- 正在理发 ---
23        // get_haircut();
24    } else {      // 没有空椅子了
25        V(mutex); // 必须释放锁再离开
26        
27        // --- 顾客离开 ---
28        leave();
29    }
30}
31
32// 理发师进程-消费者
33Barber() {
34    while(true) {
35        P(customers); // 等待顾客到来. 如果没有顾客, 则在椅子上睡眠. 
36
37        // 下述三行其实就是生产者-消费者模型中的 V(empty) 操作, 因为顾客坐下之后, 等待用的椅子就多了一把, 所以 empty 信号量加一
38        P(mutex);
39        waiting_chairs++; // 一个顾客从等待区坐到理发椅上, 空出一把椅子
40        V(mutex);
41
42        // 如果是生产者-消费者模型, 这里就可以结束了, 因为使用完物品就可以结束了
43
44        V(barbers); // 我这个理发师被叫醒了, 可以开始理发了
45
46        // --- 理发进行中 ---
47        cut_hair();
48        // --- 理发结束 ---
49    }
50}
